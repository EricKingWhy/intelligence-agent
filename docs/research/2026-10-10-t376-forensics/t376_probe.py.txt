"""#376 forensics probe (env-gated pytest plugin, loaded via `pytest -p t376_probe`).

Goal: name the holder of a memory-v2 SQLite write lock, or produce the wall-clock
distribution that lets us decide "environment item".

Mechanism: every aiosqlite queued call funnels through ``Connection._execute(fn, *args)``
(aiosqlite/core.py:150) -- execute/commit/rollback/close all go through it. We wrap
``_execute`` to bracket each ``BEGIN IMMEDIATE`` with its matching COMMIT / ROLLBACK /
close and measure how long the write transaction stayed open. Three records:

1. ``write_hold``               -- every completed BEGIN IMMEDIATE hold (ms). The
                                   distribution gives a wall-clock quantity (p50/p99/max)
                                   for "normal" holds.
2. ``long_write_hold``          -- a hold >= ``T376_PROBE_HOLD_MS`` (default 200), with the
                                   stack captured at BEGIN (names the call site for the holder).
3. watchdog (thread)            -- every ``T376_PROBE_POLL_MS`` opens a *fresh* raw sqlite3
                                   connection to each DB that has a live tracked aiosqlite
                                   connection, sets ``busy_timeout=1`` and tries
                                   ``BEGIN IMMEDIATE``. A SQLITE_BUSY means a write lock is
                                   held *right now* -> emit ``write_lock_busy_detected`` with a
                                   snapshot of every live holder from (1). If the snapshot is
                                   empty, the lock is held by a raw sqlite3 / external holder.

Enable: ``T376_PROBE=1`` [``T376_PROBE_OUT=<path.jsonl>``] [``T376_PROBE_HOLD_MS=200``]
[``T376_PROBE_FLOOR_MS=1``] [``T376_PROBE_POLL_MS=50``]. Default no-op when unset.
"""

from __future__ import annotations

import atexit
import json
import os
import sqlite3
import threading
import time
import traceback

_ENABLED = os.environ.get("T376_PROBE") == "1"

if _ENABLED:
    import asyncio

    import aiosqlite

    _OUT = os.environ.get("T376_PROBE_OUT", "/tmp/t376_probe.jsonl")
    _HOLD_MS = int(os.environ.get("T376_PROBE_HOLD_MS", "200"))
    _FLOOR_MS = int(os.environ.get("T376_PROBE_FLOOR_MS", "1"))
    _POLL_MS = int(os.environ.get("T376_PROBE_POLL_MS", "50"))

    _lock = threading.Lock()
    _holders: dict[int, dict] = {}    # id(conn) -> {path, stack, t0, thread, task}
    _paths: dict[int, str] = {}       # id(conn) -> db path (live aiosqlite conns)
    _open_count: dict[str, int] = {}  # path -> currently-open aiosqlite conns
    _out_lock = threading.Lock()
    _stats = {"holds": 0, "long": 0, "busy": 0}

    _T0 = time.time()

    def _emit(event: str, **fields) -> None:
        rec = {"ts": round(time.time(), 3), "t": round(time.time() - _T0, 3),
               "pid": os.getpid(), "event": event, **fields}
        line = json.dumps(rec, ensure_ascii=False)
        with _out_lock:
            with open(_OUT, "a", encoding="utf-8") as fh:
                fh.write(line + "\n")
        print(f"[t376_probe] {line}", flush=True)

    def _task_name() -> str:
        try:
            task = asyncio.current_task()
        except RuntimeError:
            return "no-loop"
        if task is None:
            return "no-task"
        return task.get_name() if hasattr(task, "get_name") else repr(task)

    _orig_connect = aiosqlite.connect

    def _connect(*args, **kwargs):
        conn = _orig_connect(*args, **kwargs)
        path = str(args[0]) if args else str(kwargs.get("database"))
        with _lock:
            _paths[id(conn)] = path
            _open_count[path] = _open_count.get(path, 0) + 1
        return conn

    _orig_execute_inner = aiosqlite.Connection._execute

    async def _execute(self, fn, *args, **kwargs):
        # All queued work funnels through Connection._execute(fn, *args). For DML, args[0]
        # is the SQL string; commit/rollback/close pass the bound sqlite3 method with no args.
        fn_name = getattr(fn, "__name__", "")
        sql = args[0] if args and isinstance(args[0], str) else ""
        low = sql.strip().lower() if sql else ""
        is_begin = low.startswith("begin immediate")
        is_end = low.startswith(("commit", "rollback")) or fn_name in ("commit", "rollback")
        if is_begin:
            with _lock:
                _holders[id(self)] = {
                    "path": _paths.get(id(self), "?"),
                    "stack": "".join(traceback.format_stack()),
                    "t0": time.time(),
                    "thread": threading.current_thread().name,
                    "task": _task_name(),
                }
        result = await _orig_execute_inner(self, fn, *args, **kwargs)
        if is_end or fn_name == "close":
            with _lock:
                h = _holders.pop(id(self), None)
            if h is not None:
                _maybe_log_hold(h, time.time())
        return result

    _orig_close = aiosqlite.Connection.close

    async def _close(self, *args, **kwargs):
        with _lock:
            h = _holders.pop(id(self), None)
            path = _paths.pop(id(self), None)
            if path is not None:
                _open_count[path] = _open_count.get(path, 1) - 1
        if h is not None:
            _maybe_log_hold(h, time.time())
        return await _orig_close(self, *args, **kwargs)

    def _maybe_log_hold(h: dict, t1: float) -> None:
        dur_ms = (t1 - h["t0"]) * 1000.0
        _stats["holds"] += 1
        if dur_ms < _FLOOR_MS:
            return
        rec = {"duration_ms": round(dur_ms, 1), "path": h["path"], "thread": h["thread"],
               "task": h["task"], "stack": h["stack"]}
        _emit("write_hold", **rec)
        if dur_ms >= _HOLD_MS:
            _stats["long"] += 1
            _emit("long_write_hold", **rec)

    def _watchdog() -> None:
        busy_sig = None
        while True:
            # Non-invasive: only open a witness connection while a tracked BEGIN IMMEDIATE
            # is actually in flight. Opening extra connections when no writer is active
            # perturbs the system under test -- e.g. `PRAGMA journal_mode=WAL` fails with
            # SQLITE_BUSY when other connections have the database open.
            with _lock:
                paths = sorted({h["path"] for h in _holders.values() if h["path"] != "?"})
            for path in paths:
                try:
                    probe = sqlite3.connect(path, timeout=0.05)
                except Exception:
                    continue
                try:
                    probe.execute("PRAGMA busy_timeout=1")
                    try:
                        probe.execute("BEGIN IMMEDIATE")
                    except sqlite3.OperationalError as err:
                        msg = str(err).lower()
                        if "locked" in msg or "busy" in msg:
                            with _lock:
                                snap = [
                                    {"path": h["path"],
                                     "held_ms": round((time.time() - h["t0"]) * 1000, 1),
                                     "thread": h["thread"], "task": h["task"],
                                     "stack": h["stack"]}
                                    for h in _holders.values()
                                ]
                            # dedup repeats of the same held-second bucket for readability
                            sig = tuple(sorted((h["path"], round(h["held_ms"] / 1000))
                                               for h in snap))
                            if sig != busy_sig:
                                busy_sig = sig
                                _stats["busy"] += 1
                                _emit("write_lock_busy_detected", probe_path=path,
                                      holders=snap, holders_count=len(snap))
                    else:
                        busy_sig = None
                        try:
                            probe.execute("ROLLBACK")
                        except sqlite3.Error:
                            pass
                except sqlite3.Error:
                    pass
                finally:
                    probe.close()
            time.sleep(_POLL_MS / 1000.0)

    def _summary() -> None:
        try:
            with open(_OUT, "a", encoding="utf-8") as fh:
                fh.write(json.dumps({"ts": round(time.time(), 3), "pid": os.getpid(),
                                     "event": "probe_summary", **_stats}) + "\n")
        except Exception:
            pass

    aiosqlite.connect = _connect
    aiosqlite.Connection._execute = _execute
    aiosqlite.Connection.close = _close
    threading.Thread(target=_watchdog, name="t376-probe-watchdog", daemon=True).start()
    atexit.register(_summary)
    _emit("probe_armed", hold_ms=_HOLD_MS, floor_ms=_FLOOR_MS, poll_ms=_POLL_MS, out=_OUT)
